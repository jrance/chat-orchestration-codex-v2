## 2) Engine wiring

**Goal:** Runtime should actually **execute** the concurrent plan and append the merged/selected text to `state.messages` while emitting telemetry.

**File:** `app/runtime/engine.py`

```python
from __future__ import annotations
import asyncio
from typing import Any, Dict
from app.runtime.patterns.concurrent import run_concurrent
from app.telemetry.bus import publish_telemetry

async def _run_child_adapter(run_ctx: Dict[str, Any], child_id: str, timeout_s: int) -> dict:
    """Adapter the concurrent executor uses to run a single child node and return a ChildResult."""
    label = run_ctx["plan"]["nodes_map"][child_id].get("label", child_id)

    try:
        publish_telemetry(
            {"type":"telemetry.concurrent.child.start","nodeId":child_id,"label":label},
            run_ctx["run_id"]
        )

        async def _do():
            # Run one child; you likely already have this function:
            #   await run_node_once(run_ctx, child_id)
            before = list(run_ctx["state"].get("messages", []))
            await run_node_once(run_ctx, child_id)  # ← your existing per-node runner
            after = run_ctx["state"].get("messages", [])

            # Capture the newest assistant text produced by this child
            new_text = None
            for m in reversed(after):
                if m not in before and m.get("role") == "assistant":
                    new_text = m.get("content")
                    break

            return {
                "nodeId": child_id,
                "label": label,
                "status": "completed",
                "output_text": new_text,
                "confidence": None,
                "usage": None,
                "meta": {},
            }

        res = await asyncio.wait_for(_do(), timeout=timeout_s)
        publish_telemetry(
            {"type":"telemetry.concurrent.child.end","nodeId":child_id,"label":label},
            run_ctx["run_id"]
        )
        return res

    except asyncio.TimeoutError:
        publish_telemetry(
            {"type":"telemetry.concurrent.child.timeout","nodeId":child_id,"label":label},
            run_ctx["run_id"]
        )
        return {"nodeId": child_id, "label": label, "status":"timeout", "output_text": None,
                "confidence": None, "usage": None, "meta": {}}

    except asyncio.CancelledError:
        publish_telemetry(
            {"type":"telemetry.concurrent.child.cancelled","nodeId":child_id,"label":label},
            run_ctx["run_id"]
        )
        return {"nodeId": child_id, "label": label, "status":"cancelled", "output_text": None,
                "confidence": None, "usage": None, "meta": {}}

    except Exception as e:
        publish_telemetry(
            {"type":"telemetry.concurrent.child.error","nodeId":child_id,"label":label,"error":str(e)},
            run_ctx["run_id"]
        )
        return {"nodeId": child_id, "label": label, "status":"error", "error": str(e),
                "output_text": None, "confidence": None, "usage": None, "meta": {}}

async def run_concurrent_node(run_ctx: Dict[str, Any], node_plan: Dict[str, Any]) -> Dict[str, Any]:
    """Invoke concurrent executor and append final text to state.messages."""
    cfg = node_plan["cfg"]
    children_ids = node_plan["children"]

    # Build descriptors for the executor
    children = [{"id": cid, "label": run_ctx["plan"]["nodes_map"][cid].get("label", cid)}
                for cid in children_ids]

    exec_ctx = {
        "run_id": run_ctx["run_id"],
        "plan": run_ctx["plan"],
        "state": run_ctx["state"],
        "run_child": lambda child: _run_child_adapter(run_ctx, child["id"], cfg["timeout_seconds"]),
    }

    result = await run_concurrent(children, cfg, exec_ctx)

    # Determine final text
    if result["strategy"] == "Synthesize":
        final_text = result.get("merged_text") or ""
    else:
        chosen = result.get("chosen") or {}
        final_text = chosen.get("output_text") or ""

    if final_text:
        msgs = list(run_ctx["state"].get("messages") or [])
        msgs.append({"role":"assistant","content": final_text})
        run_ctx["state"]["messages"] = msgs

    publish_telemetry({
        "type":"telemetry.concurrent.merge.decision",
        "strategy": result["strategy"],
        "chosenNodeId": (result.get("chosen") or {}).get("nodeId"),
        "childrenCount": len(result.get("children") or []),
    }, run_ctx["run_id"])

    return run_ctx["state"]
```

## 3) Streaming (SSE) behavior

- **Do NOT** interleave child token deltas into the main **Responses** stream.
- While children run, emit **telemetry‑only** events for child deltas (e.g., `telemetry.llm.delta`, labeled by `nodeId`/`label`).
- When merge decides, **only then** stream the final text on the **Responses** stream:
  - `Synthesize`: stream the synthesized text (or send in a single chunk if non‑streaming).
  - `FirstBest` / `HighestScore`: stream the chosen child’s output.
- Always end with `response.completed`, optionally including `metadata.concurrent`:
  ```json
  {
    "type": "response.completed",
    "response": {
      "id": "resp_...",
      "metadata": {
        "concurrent": {
          "strategy": "HighestScore",
          "childrenCount": 3,
          "chosenNodeId": "child-abc"
        }
      }
    }
  }
  ```
